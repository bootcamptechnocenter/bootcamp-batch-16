  var cur=0, tot=contents.length+1;
  function go(i){
    cur=i;
    document.getElementById("progressBar").style.width=((i+1)/tot*100)+"%";
    document.getElementById("slideCounter").textContent=(i+1)+" / "+tot;
    document.getElementById("pageInfo").textContent=(i+1)+" / "+tot;

    document.getElementById("btnNext").disabled=(i===tot-1);

      document.getElementById("dayBadge").textContent="IDMS Training";

      document.getElementById("content").innerHTML=coverFn();

      document.querySelectorAll("#content pre code").forEach(function(el){hljs.highlightElement(el);});
    }
  }
  document.getElementById("btnPrev").addEventListener("click",function(){if(cur>0)go(cur-1);});
  document.getElementById("btnNext").addEventListener("click",function(){if(cur<tot-1)go(cur+1);});
  document.addEventListener("keydown",function(e){
    if(e.key==="ArrowRight"||e.key==="PageDown"){if(cur<tot-1)go(cur+1);}
